cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2 CACHE STRING "Picolino requires the Raspberry Pi Pico 2")
if(NOT PICO_BOARD STREQUAL "pico2")
    message(FATAL_ERROR "This initial Picolino target requires PICO_BOARD=pico2")
endif()
if(NOT DEFINED PICO_SDK_PATH)
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to a Pico SDK 2.2.0 checkout with submodules")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(picolino_pico2 C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
if(NOT PICO_RP2350)
    message(FATAL_ERROR "Picolino must be compiled for RP2350")
endif()
add_executable(picolino_bus_monitor bus_monitor.c)
target_include_directories(picolino_bus_monitor PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(picolino_bus_monitor pico_stdlib hardware_clocks)
target_compile_options(picolino_bus_monitor PRIVATE -Wall -Wextra -Werror)
# GPIO0/1 are CPC address inputs, so never enable UART stdio.
pico_enable_stdio_uart(picolino_bus_monitor 0)
pico_enable_stdio_usb(picolino_bus_monitor 1)
pico_set_program_name(picolino_bus_monitor "Picolino bus monitor")
pico_set_program_version(picolino_bus_monitor "0.1.0")
pico_add_extra_outputs(picolino_bus_monitor)
